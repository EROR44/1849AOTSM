		context_type = player_context
		window_name = "aotsm_welcome_screen_container"

		ai_enabled = {
			always = no
		}

		visible = {
			is_ai = no
			has_country_flag = aotsm_open_welcome_screen_flag
		}

		triggers = {

		}

		effects = {
			close_button_click = {
				clr_country_flag = aotsm_open_welcome_screen_flag
			}

			aotsm_welcome_screen_others_button_click = {
				aotsm_welcome_initialize_buttons = yes
				if = {
					limit = {
						check_variable = {
							ROOT.aotsm_others_page_open = 0
						}
					}
					set_variable = { ROOT.aotsm_others_page_open = 1 }
					set_variable = { ROOT.aotsm_others_page_num = 1 }
				}
				else = {
					set_variable = { ROOT.aotsm_others_page_open = 0 }
				}
			}

		}
	}

	aotsm_welcome_screen_buttons_left = {

					limit = {
						tag = AZT
					}

				}
				else_if = {
					limit = {
						tag = CBI
					}

				}
				else_if = {
					limit = {
						tag = PRT
					}
					set_variable = { ROOT.aotsm_lore_section_num = 3 }
				}
				else_if = {
					limit = {
						tag = PRU
					}
					set_variable = { ROOT.aotsm_lore_section_num = 4 }
				}
				else_if = {
					limit = {
						tag = BYZ
					}
					set_variable = { ROOT.aotsm_lore_section_num = 5 }
				}
				else_if = {
					limit = {
						tag = RUS
					}
					set_variable = { ROOT.aotsm_lore_section_num = 6 }
				}
				else_if = {
					limit = {
						tag = FRR
					}
					set_variable = { ROOT.aotsm_lore_section_num = 7 }
				}
				else_if = {
					limit = {
						tag = QIN
					}
					set_variable = { ROOT.aotsm_lore_section_num = 8 }
				}
				else_if = {
					limit = {
						tag = JAP
					}
					set_variable = { ROOT.aotsm_lore_section_num = 9 }
				}

					set_variable = { ROOT.aotsm_lore_section_num = 100 } # No information about history of this country
				}

			}

			}
