	aotsm_welcome_screen = {
		context_type = player_context
		window_name = "aotsm_welcome_screen_container"

		ai_enabled = {
			always = no
		}

		visible = {
			is_ai = no
			has_country_flag = aotsm_open_welcome_screen_flag
		}

		triggers = {
			
		}

		effects = {
			close_button_click = {
				clr_country_flag = aotsm_open_welcome_screen_flag
			}
		
			aotsm_welcome_screen_others_button_click = {
				if = {
					limit = {
						check_variable = {
							ROOT.aotsm_others_page_open = 0
						}
					}
					set_variable = { ROOT.aotsm_others_page_open = 1 }
					set_variable = { ROOT.aotsm_others_page_num = 1 }
				}
				else = {
					set_variable = { ROOT.aotsm_others_page_open = 0 }
				}
			}

		}
	}
