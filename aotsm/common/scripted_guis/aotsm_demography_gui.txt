scripted_gui = {
	aotsm_demo_toggle = {
		context_type = selected_state_context
		parent_window_token = selected_state_view
		window_name = "aotsm_demo_toggle_window"
		ai_enabled = { always = no }
		effects = {
			aotsm_demo_toggle_button_click = {
				ROOT = {
					if = {
						limit = { has_country_flag = aotsm_demo_panel_open }
						clr_country_flag = aotsm_demo_panel_open
					}
					else = { set_country_flag = aotsm_demo_panel_open }
				}
			}
		}
	}
	aotsm_demo_panel = {
		context_type = selected_state_context
		parent_window_token = selected_state_view
		window_name = "aotsm_demo_state_panel"
		visible = { ROOT = { has_country_flag = aotsm_demo_panel_open } }
		ai_enabled = { always = no }
		effects = { aotsm_demo_fold_button_click = { ROOT = { clr_country_flag = aotsm_demo_panel_open } } }
		dynamic_lists = {
			aotsm_demo_culture_grid = {
				array = aotsm_demo_c_ids
				value = aotsm_demo_c_entry
				change_scope = no
				entry_container = "aotsm_demo_culture_row"
			}
			aotsm_demo_religion_grid = {
				array = aotsm_demo_r_ids
				value = aotsm_demo_r_entry
				change_scope = no
				entry_container = "aotsm_demo_religion_row"
			}
		}
	}
}
