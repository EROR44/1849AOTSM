##All Qing triggers is here

QIN_check_num_of_arsenals_under_construction_trigger = {

    custom_trigger_tooltip = {
		tooltip = QIN_num_of_building_arsenals_tt
        check_variable = {
            QIN.QIN_num_of_building_arsenals < 2
        }
    }
}

QIN_check_num_of_regions_with_recruitment_trigger = {

    custom_trigger_tooltip = {
		tooltip = QIN_num_of_regions_with_recruitment_tt
        check_variable = {
            QIN.QIN_recrut_in_process_num < 5
        }
    }
}

QIN_the_new_emperor_is_choosen_trigger = {

    custom_trigger_tooltip = {
		tooltip = QIN_the_new_emperor_is_choosen_trigger_tt
        ROOT = {
            OR = {
                has_country_flag = QIN_ichzu_is_new_emperror_flag
                has_country_flag = QIN_isin_is_new_emperror_flag
                has_country_flag = QIN_isuan_is_new_emperror_flag
            }
        }
    }
}
