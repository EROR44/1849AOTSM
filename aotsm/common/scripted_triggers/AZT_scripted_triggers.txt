AZT_the_division_limit_will_not_be_reached_after_the_draft_of_the_militia = {
    custom_trigger_tooltip = {
		tooltip = AZT_the_division_limit_will_not_be_reached_after_the_draft_of_the_militia_tt
		NOT = {
			if = {
				limit = {
					has_idea = colonial_area
				}
				check_variable = { num_divisions > 9 }
			}
			else_if = {
				limit = {
					has_idea = small_colony
				}
				check_variable = { num_divisions > 19 }
			}
			else_if = {
				limit = {
					has_idea = average_colony
				}
				check_variable = { num_divisions > 29 }
			}
			else_if = {
				limit = {
					has_idea = colonial_autonomy
				}
				check_variable = { num_divisions > 49 }
			}
			else_if = {
				limit = {
					has_idea = little_country
				}
				check_variable = { num_divisions > 19 }
			}
			else_if = {
				limit = {
					has_idea = medium_country
				}
				check_variable = { num_divisions > 29 }
			}
			else_if = {
				limit = {
					has_idea = large_country
				}
				check_variable = { num_divisions > 49 }
			}
			else_if = {
				limit = {
					has_idea = potential_world_power
				}
				check_variable = { num_divisions > 99 }
			}
			else_if = {
				limit = {
					has_idea = world_power
				}
				check_variable = { num_divisions > 149 }
			}
			else = {
				check_variable = { num_divisions > 9 }
			}
		}
	}
}
