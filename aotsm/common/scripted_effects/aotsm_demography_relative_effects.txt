		aotsm_demography_relative_applied_var = 0
	}

	set_temp_variable = {
		aotsm_demography_relative_delta_var = 0
	}

	set_temp_variable = {
		aotsm_demography_relative_loop_break_var = 0
	}

	aotsm_demography_clear_relative_arrays_effect = yes

	if = {
		limit = {
			has_global_flag = aotsm_demo_initialized_v1

			is_in_array = {
				aotsm_demo_r_ids = aotsm_demography_target_id_var
			}

			NOT = {
				check_variable = {
					var = aotsm_demography_relative_percent_var
					value = 0
					compare = equals
				}
			}

			check_variable = {
				var = aotsm_demography_relative_percent_var
				value = -100
				compare = greater_than_or_equals
			}

			check_variable = {
				var = aotsm_demography_relative_percent_var
				value = 100
				compare = less_than_or_equals
			}
		}

		aotsm_demography_initialize_relative_registry_effect = yes

		for_each_loop = {
			array = aotsm_demo_r_ids
			value = aotsm_demography_relative_id_var
			index = aotsm_demography_relative_source_index_var
			break = aotsm_demography_relative_loop_break_var

			set_temp_variable = {
				aotsm_demography_relative_key_var = global.aotsm_demography_relative_religion_keys_array^aotsm_demography_relative_id_var
			}

			aotsm_demography_read_relative_group_effect = yes

			add_to_temp_array = {
				aotsm_demography_relative_ids_array = aotsm_demography_relative_id_var
			}

			add_to_temp_array = {
				aotsm_demography_relative_keys_array = aotsm_demography_relative_key_var
			}

			add_to_temp_array = {
				aotsm_demography_relative_old_shares_array = aotsm_demography_relative_share_var
			}
		}

		aotsm_demography_calculate_relative_change_effect = yes

		if = {
			limit = {
				check_variable = {
					var = aotsm_demography_relative_applied_var
					value = 1
					compare = equals
				}
			}

			clear_array = aotsm_demo_r_ids

			for_each_loop = {
				array = aotsm_demography_relative_sorted_ids_array
				value = aotsm_demography_relative_id_var
				index = aotsm_demography_relative_result_index_var
				break = aotsm_demography_relative_loop_break_var

				add_to_array = {
					aotsm_demo_r_ids = aotsm_demography_relative_id_var
				}
			}

			aotsm_demo_refresh_state = yes
			aotsm_demo_update_scroll_extents = yes

			owner = {
				aotsm_demo_refresh_country = yes
			}

			force_update_map_mode = {
				mapmode = religion_map_mode
			}
		}
	}

	aotsm_demography_clear_relative_arrays_effect = yes
