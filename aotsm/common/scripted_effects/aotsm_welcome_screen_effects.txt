# Welcome screen initialization and navigation.
aotsm_welcome_initialize_buttons = {

	add_to_array = { global.aotsm_welcome_others_button_entries = 1 } # aotsm_others_developers_button
	clear_array = global.aotsm_welcome_button_entries
	add_to_array = { global.aotsm_welcome_button_entries = 0 } # aotsm_buttons_left_common_story_button
	add_to_array = { global.aotsm_welcome_button_entries = 1 } # aotsm_buttons_left_cur_country_button
	add_to_array = { global.aotsm_welcome_button_entries = 2 } # aotsm_buttons_left_aztec_story_button
	add_to_array = { global.aotsm_welcome_button_entries = 3 } # aotsm_buttons_left_kbo_story_button
	add_to_array = { global.aotsm_welcome_button_entries = 4 } # aotsm_buttons_left_prt_story_button
	add_to_array = { global.aotsm_welcome_button_entries = 5 } # aotsm_buttons_left_pru_story_button
	add_to_array = { global.aotsm_welcome_button_entries = 6 } # aotsm_buttons_left_byz_story_button
	add_to_array = { global.aotsm_welcome_button_entries = 7 } # aotsm_buttons_left_rus_story_button
	add_to_array = { global.aotsm_welcome_button_entries = 8 } # aotsm_buttons_left_frr_story_button
	add_to_array = { global.aotsm_welcome_button_entries = 9 } # aotsm_buttons_left_qin_story_button
	add_to_array = { global.aotsm_welcome_button_entries = 10 } # aotsm_buttons_left_jap_story_button
}

aotsm_welcome_on_startup = {
	aotsm_welcome_initialize_buttons = yes
	every_country = { #welcome screen GUI block
		limit = {
			is_ai = no
		}
		#country_event = history.1 #потом убрать
		set_country_flag = aotsm_open_welcome_screen_flag
		set_variable = {
			aotsm_contry_lore_page = 0
		}
		set_variable = {
			aotsm_lore_section_num = 0
		}
		set_variable = {
			aotsm_others_section_num = 0
		}
		set_variable = {
			aotsm_others_page_num  = 0
		}
	}
}

# Page counts are maintained here, independently of navigation buttons.
aotsm_welcome_read_page = {
	set_temp_variable = { aotsm_welcome_page_count = 1 }
	if = {
		limit = { check_variable = { aotsm_others_page_open = 1 } }
		set_temp_variable = { aotsm_welcome_current_page = aotsm_others_page_num }
		if = {
			limit = { check_variable = { aotsm_others_section_num = 0 } }
			set_temp_variable = { aotsm_welcome_page_count = 2 }
		}
	}
	else = {
		set_temp_variable = { aotsm_welcome_current_page = aotsm_country_lore_page_num }
		if = {
			limit = { check_variable = { aotsm_lore_section_num = 1 } }
			set_temp_variable = { aotsm_welcome_page_count = 5 }
		}
		else_if = {
			limit = { check_variable = { aotsm_lore_section_num = 2 } }
			set_temp_variable = { aotsm_welcome_page_count = 5 }
		}
		else_if = {
			limit = { check_variable = { aotsm_lore_section_num = 3 } }
			set_temp_variable = { aotsm_welcome_page_count = 5 }
		}
		else_if = {
			limit = { check_variable = { aotsm_lore_section_num = 4 } }
			set_temp_variable = { aotsm_welcome_page_count = 3 }
		}
		else_if = {
			limit = { check_variable = { aotsm_lore_section_num = 5 } }
			set_temp_variable = { aotsm_welcome_page_count = 8 }
		}
		else_if = {
			limit = { check_variable = { aotsm_lore_section_num = 6 } }
			set_temp_variable = { aotsm_welcome_page_count = 2 }
		}
		else_if = {
			limit = { check_variable = { aotsm_lore_section_num = 7 } }
			set_temp_variable = { aotsm_welcome_page_count = 3 }
		}
		else_if = {
			limit = { check_variable = { aotsm_lore_section_num = 8 } }
			set_temp_variable = { aotsm_welcome_page_count = 7 }
		}
		else_if = {
			limit = { check_variable = { aotsm_lore_section_num = 9 } }
			set_temp_variable = { aotsm_welcome_page_count = 5 }
		}
	}
}

aotsm_welcome_write_page = {
	if = {
		limit = { check_variable = { aotsm_others_page_open = 1 } }
		set_variable = { aotsm_others_page_num = aotsm_welcome_current_page }
	}
	else = {
		set_variable = { aotsm_country_lore_page_num = aotsm_welcome_current_page }
	}
}

aotsm_welcome_next_page = {
	aotsm_welcome_read_page = yes
	add_to_temp_variable = { aotsm_welcome_current_page = 1 }
	if = {
		limit = { check_variable = { var = aotsm_welcome_current_page value = aotsm_welcome_page_count compare = greater_than } }
		set_temp_variable = { aotsm_welcome_current_page = 1 }
	}
	else_if = {
		limit = { check_variable = { var = aotsm_welcome_current_page value = 1 compare = less_than } }
		set_temp_variable = { aotsm_welcome_current_page = aotsm_welcome_page_count }
	}
	aotsm_welcome_write_page = yes
}

aotsm_welcome_prev_page = {
	aotsm_welcome_read_page = yes
	add_to_temp_variable = { aotsm_welcome_current_page = -1 }
	if = {
		limit = { check_variable = { var = aotsm_welcome_current_page value = aotsm_welcome_page_count compare = greater_than } }
		set_temp_variable = { aotsm_welcome_current_page = 1 }
	}
	else_if = {
		limit = { check_variable = { var = aotsm_welcome_current_page value = 1 compare = less_than } }
		set_temp_variable = { aotsm_welcome_current_page = aotsm_welcome_page_count }
	}
	aotsm_welcome_write_page = yes
}
