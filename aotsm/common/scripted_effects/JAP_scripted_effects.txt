JAP_check_owned_states_array_effect = {
	clear_array = JAP.aotsm_owned_states_array

	every_state = {
		limit = {
			is_owned_and_controlled_by = JAP    
		}
		add_to_array = {
			array = JAP.aotsm_owned_states_array
			value = THIS #THIS означает текущий стейт
		}
	}
}

#SWAP_PARAMETERS

JAP_go_to_war_economy_effect = {
    if = {
        limit = {
            has_idea = partial_economic_mobilisation
        }
        swap_ideas = {
            remove_idea = partial_economic_mobilisation
            add_idea = war_economy
        }
    }
    else_if = {
        limit = {
            has_idea = low_economic_mobilisation
        }
        swap_ideas = {
            remove_idea = low_economic_mobilisation
            add_idea = war_economy
        }
    }
    else_if = {
        limit = {
            has_idea = civilian_economy
        }
        swap_ideas = {
            remove_idea = civilian_economy
            add_idea = war_economy
        }
    }
}

JAP_calculate_factions_superiority_num_effect = {
	set_variable = {
		jap_cw_total_controlled_states_num = {
			value = JAP.num_of_controlled_states
			add = JAS.num_of_controlled_states
			add = TUK.num_of_controlled_states
		}
	}

	set_variable = {
		JAP.jap_cw_num_of_controlled_states = {
			value = JAP.num_of_controlled_states
			multiply = 100
			divide = jap_cw_total_controlled_states_num
		}
	}

	set_variable = {
		JAS.jap_cw_num_of_controlled_states = {
			value = JAS.num_of_controlled_states
			multiply = 100
			divide = jap_cw_total_controlled_states_num
		}
	}

	set_variable = {
		TUK.jap_cw_num_of_controlled_states = {
			value = TUK.num_of_controlled_states
			multiply = 100
			divide = jap_cw_total_controlled_states_num
		}
	}
}

JAP_cw_who_is_winner_effect = {
    if = {
        limit = {
            check_variable = { JAP.jap_cw_num_of_controlled_states > JAS.jap_cw_num_of_controlled_states }
            check_variable = { JAP.jap_cw_num_of_controlled_states > TUK.jap_cw_num_of_controlled_states }
        }
        JAP = {
            set_country_flag = JAP_is_winner_in_civil_war_flag
        }
    }
    else_if = {
        limit = {
            check_variable = { JAS.jap_cw_num_of_controlled_states > JAP.jap_cw_num_of_controlled_states }
            check_variable = { JAS.jap_cw_num_of_controlled_states > TUK.jap_cw_num_of_controlled_states }
        }
        JAS = {
            set_country_flag = JAS_is_winner_in_civil_war_flag
        }
    }
    else_if = {
        limit = {
            check_variable = { TUK.jap_cw_num_of_controlled_states > JAP.jap_cw_num_of_controlled_states }
            check_variable = { TUK.jap_cw_num_of_controlled_states > JAS.jap_cw_num_of_controlled_states }
        }
        TUK = {
            set_country_flag = TUK_is_winner_in_civil_war_flag
        }
    }
}
