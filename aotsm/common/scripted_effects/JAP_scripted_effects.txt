
#STATE ARRAYS

JAP_check_owned_states_array_effect = {
	clear_array = JAP.aotsm_owned_states_array

	every_state = {
		limit = {
			is_owned_and_controlled_by = JAP    
		}
		add_to_array = {
			array = JAP.aotsm_owned_states_array
			value = THIS #THIS означает текущий стейт
		}
	}
}

#SWAP_PARAMETERS

JAP_go_to_war_economy_effect = {
    if = {
        limit = {
            has_idea = partial_economic_mobilisation
        }
        swap_ideas = {
            remove_idea = partial_economic_mobilisation
            add_idea = war_economy
        }
    }
    else_if = {
        limit = {
            has_idea = low_economic_mobilisation
        }
        swap_ideas = {
            remove_idea = low_economic_mobilisation
            add_idea = war_economy
        }
    }
    else_if = {
        limit = {
            has_idea = civilian_economy
        }
        swap_ideas = {
            remove_idea = civilian_economy
            add_idea = war_economy
        }
    }
}
