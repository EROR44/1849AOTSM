AZT_austrian_trade_monopoly_effect = {
	every_other_country = {
       	remove_opinion_modifier = {
        	target = AZT
        	modifier = end_trade
    	}
    }
	if = {
		limit = {
			has_completed_focus = AZT_trade_with_japan
		}
		every_other_country = {
			limit = {
				NOT = { tag = MEC }
				NOT = { tag = JAP }
				NOT = { tag = TUK }
				NOT = { 
					OR = {
						is_puppet_of = JAP
						is_subject_of = JAP
						is_in_faction_with = JAP
					}
				}
			}
			add_opinion_modifier = {
        		target = AZT
            	modifier = end_trade
        	}
		}
	}
	else_if = {
		limit = {
			has_completed_focus = AZT_trade_with_sonora
			NOT = {
				SON = {
					is_puppet_of = PRT
					is_subject_of = PRT
				}
			}
		}
		every_other_country = {
			limit = {
				NOT = { tag = MEC }
				NOT = { tag = SON }
				NOT = {
					OR = {
						is_puppet_of = SON
						is_subject_of = SON
					}
				}
			}
			add_opinion_modifier = {
        		target = AZT
            	modifier = end_trade
        	}
		}
	}
	else_if = {
		limit = {
			has_completed_focus = AZT_trade_with_sonora
			SON = {
				is_puppet_of = PRT
				is_subject_of = PRT
			}
		}
		every_other_country = {
			limit = {
				NOT = { tag = MEC }
				NOT = { tag = SON }
				NOT = {
					OR = {
						is_puppet_of = SON
						is_subject_of = SON
					}
				}
				NOT = { tag = PRT }
				NOT = {
					OR = {
						is_puppet_of = PRT
						is_subject_of = PRT
					}
				}
			}
			add_opinion_modifier = {
        		target = AZT
            	modifier = end_trade
        	}
		}
	}
	else = {
		every_other_country = {
			limit = {
				NOT = { tag = MEC }
			}
       		add_opinion_modifier = {
        		target = AZT
            	modifier = end_trade
        	}
    	}
	}
}
