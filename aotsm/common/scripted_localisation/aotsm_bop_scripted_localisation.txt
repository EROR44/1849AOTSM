#Change of BOP icons is scripted here

defined_text = {
	name = aotsm_get_jap_bop_iaf_icon
	text = {
		trigger = {
			tag = JAP
			check_variable = { 
				JAP_bop_iaf_icon_num = 0
			}
		}
		localization_key = iaf_icon_1
	}
	text = {
		trigger = {
			tag = JAP
			check_variable = { 
				JAP_bop_iaf_icon_num = 1
			}
		}
		localization_key = iaf_icon_2
	}
	text = {
		trigger = {
			tag = JAP
			check_variable = { 
				JAP_bop_iaf_icon_num = 2
			}
		}
		localization_key = iaf_icon_3
	}
}
