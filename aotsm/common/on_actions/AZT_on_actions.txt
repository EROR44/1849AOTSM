on_actions = {

    on_startup = {
        effect = {
            AZT = { country_event = { id = azt_scripted_events.2 days = 13 random_hours = 4 } }
        }
    }

    on_weekly_AZT = {
        effect = {
            if = {
                limit = {
                    has_idea = AZT_aus_trade_monopoly_idea
                }
                AZT_austrian_trade_monopoly_effect = yes
            }
        }
    }

    on_peace = {
		effect = {
			# Aztec switch law
			if = {
				limit = {
                    tag = AZT
					has_idea = AZT_macehualtin_law_war
				}
				swap_ideas = {
                    remove_idea = AZT_macehualtin_law_war
				    add_idea = AZT_macehualtin_law_peacetime
                }
                multiply_variable = { AZT.get_cost_for_check_the_readness_of_the_militia = 3 } # Decision Cost
                delete_units = { 
				    division_template = "Macehualtin"
				    disband = no # if yes, equipment will be returned to country equipment. default is no
			    }
			}
        }
    }

    on_war = {
        effect = {
            # Aztec switch law
            if = {
                limit = {
                    tag = AZT
                    AZT = { has_idea = AZT_macehualtin_law_peacetime }
                }
                AZT = {
                    swap_ideas = {
                        remove_idea = AZT_macehualtin_law_peacetime
                        add_idea = AZT_macehualtin_law_war
                    }
                }
               divide_variable = { AZT.get_cost_for_check_the_readness_of_the_militia = 3 } # Decision Cost
            }
        }
    }

    on_monthly_AZT = {
        effect = {
            if = {
                limit = {
                    has_country_flag = AZT_cholera_gui_visible_flag
                }
                AZT_get_random_cholera_event_effect = yes
            }
        }
    }
}
