on_actions = {

    on_weekly_AZT = {
        effect = {
            if = {
                limit = {
                    has_idea = AZT_aus_trade_monopoly_idea
                }
                AZT_austrian_trade_monopoly_effect = yes
            }
            if = {
                limit = {
                    has_country_flag = AZT_tension_gui_visible_flag
                }
                AZT_clamp_tension_value_effect = yes
            }
        }
    }

    on_peace = {
		effect = {
			# Aztec switch law
			if = {
				limit = {
                    tag = AZT
					has_idea = AZT_macehualtin_law_war
				}
				swap_ideas = {
                    remove_idea = AZT_macehualtin_law_war
				    add_idea = AZT_macehualtin_law_peacetime
                }
                multiply_variable = { AZT.get_cost_for_check_the_readness_of_the_militia = 3 } # Decision Cost
                delete_units = { 
				    division_template = "Macehualtin"
				    disband = no # if yes, equipment will be returned to country equipment. default is no
			    }
			}
        }
    }

    on_war = {
        effect = {
            # Aztec switch law
            if = {
                limit = {
                    tag = AZT
                    has_idea = AZT_macehualtin_law_peacetime
                }
                swap_ideas = {
                    remove_idea = AZT_macehualtin_law_peacetime
                    add_idea = AZT_macehualtin_law_war
                }
               divide_variable = { AZT.get_cost_for_check_the_readness_of_the_militia = 3 } # Decision Cost
            }
        }
    }

    on_monthly_AZT = {
        effect = {
            if = {
                limit = {
                    has_country_flag = AZT_cholera_gui_visible_flag
                }
                AZT_get_random_cholera_event_effect = yes
            }
            if = {
                limit = {
                    has_country_flag = AZT_tension_gui_visible_flag
                    has_variable = AZT_tension_growing_var
                }
                AZT_clamp_tension_value_effect = yes
                add_to_variable = { AZT_tension_value = AZT_tension_growing_var }
            }
            if = {
                limit = {
                    OR = {
                        has_idea = AZT_macehualtin_law_war
                        has_idea = AZT_macehualtin_law_peacetime
                    }
                }
                AZT_check_owned_states_array_effect = yes
            }
        }
    }

    #ROOT is winner, FROM is loser
	on_before_peace_conference_start = {
		# DEBUG - where we haven't scripted peace
		effect = {
			if = {
				limit = {
					FROM = { tag = AZT }  # Aztec is loser
				}
				log = "[FROM.GetName] lost against [ROOT.GetName]"
			}
		}
	}

    on_state_control_changed = {
        effect = {
            if = {
                limit = {
                    FROM = {
                        tag = AZT
                        has_country_flag = AZT_third_maya_aztec_war_flag
                    }
                    ROOT = {
                        tag = MAY
                    }
                    FROM.FROM = {
                        is_owned_by = AZT
                    }
                }
                aztec_lose_in_third_maya_aztec_war = yes
            }
        }
    }
}
