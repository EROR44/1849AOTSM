	is_triggered_only = yes
	fire_only_once = no

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event aotsm_demography_test.1"
	}

	option = {
		name = aotsm_demography_test_1_1_t

		trigger = {
			has_global_flag = aotsm_demo_initialized_v1
			capital_scope = {
				check_variable = {
					var = aotsm_demo_c_largest
					value = 0
					compare = greater_than
				}

				check_variable = {
					var = aotsm_demo_c_largest
					value = 100
					compare = less_than
				}
			}
		}

		ai_chance = {
			factor = 0
		}

		custom_effect_tooltip = aotsm_demography_test_change_tt

		hidden_effect = {
			capital_scope = {
				set_temp_variable = {
					aotsm_demography_target_id_var = aotsm_demo_c_leader
				}
